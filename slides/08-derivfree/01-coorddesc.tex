\begin{framei}{coordinate descent}
\item Idea: instead of gradient, use coordinate directions for descent
\item No gradient needed: objective function may be non-differentiable
\comment[NOTE]{BB}{austauschen mit eigener grafik}
\vfill
\item[] \textbf{Approach:}
\item Select starting point $\xv^{[0]} = (x^{[0]}_1, \ldots, x^{[0]}_d)$
\item Step $t$: Minimize $f$ along $x_i$ for each dimension $i$ for fixed $x^{[t]}_1, \ldots,x^{[t]}_{i-1}$ and $x^{[t-1]}_{i+1}, \ldots,x^{[t-1]}_d$
\item Minimum is determined with (exact/inexact) line search
\item Order of dimensions: cyclic $1, 2, \ldots, d$, or any other permutation
\comment[NOTE]{CK}{verstehe ich nicht}
\end{framei}


\begin{framei}{convergence criteria}
\item CD stops at a point that is minimal along each coordinate axis;
is this a global minimizer?
\sourceref{https://www.stat.cmu.edu/~ryantibs/convexopt-S15/lectures/22-coord-desc.pdf}
\comment[NOTE]{UU}{Diskussion von 3 Fällen nach Ryan Tibshirani.
1. f konvex + diff'bar
2. f konvex (Gegenbeispiel)
3. f Summe aus konvexer + diff'barer Funktion und separabler Funktion mit konvexen Summanden.
https://www.stat.cmu.edu/~ryantibs/convexopt-S15/lectures/22-coord-desc.pdf, Folien 5--7
Nachtrag: L(x) + ||x||_1 als ML-Beispiel hinzufügen.}
\item[] \textbf{Case 1:} $f$ convex and differentiable $\Rightarrow$ \textbf{yes}
\comment[NOTE]{CK}{haben oben angenommen dass f non-differentiable}
\item[] \textbf{Case 2:} $f$ convex, not differentiable $\Rightarrow$ \textbf{no} in general (next slide)
\item[] \textbf{Case 3:} $f$ = convex differentiable + \textbf{convex separable} $\Rightarrow$ \textbf{yes}
$$f(\xv) = g(\xv) + \sum_{i=1}^d h_i(x_i)$$
with $g$ convex differentiable and $h_i$ convex, possibly nonsmooth, univariate functions
\end{framei}


\begin{framei}{example: non-differentiable function}
\item No general convergence for convex, non-differentiable functions (case 2)
\item Counterexample $f(x, y) = |x + y| + 3|y - x|$, minimum at $(0, 0)$
\item At the red point, a step along either axis increases $f$,
although both steps together would bring us closer to the optimum
\item CD gets stuck at a point that is not the minimum
\imageC[0.35][https://en.wikipedia.org/wiki/Coordinate_descent]{figure_man/Nonsmooth_coordinate_descent.png}
\end{framei}


\begin{framei}{example: linear regression}
\item Minimize LM with L2 loss via CD (case 1):
$$\min g(\thetav) = \min_{\thetav} \frac{1}{2}\sumin (\yi - \thetav^T \xi)^2 = \min_{\thetav} \frac{1}{2}||\yv - \Xmat \thetav||^2 $$
where $\yv \in \R^n$, $\Xmat \in \R^{n \times d}$ with columns $\xv_1, \ldots, \xv_d \in \R^n$\\
{\footnotesize Note: smooth toy example where the minimizer along each axis is computed in closed form}
\item Assume orthogonal design for intuition, i.e., $\Xmat^T \Xmat = \id$:
\begin{align*}
g(\thetav) &= \frac{1}{2}\yv^T \yv + \frac{1}{2}\thetav^T \thetav - \yv^T \Xmat \thetav  \\
&\overset{(*)}{=} \frac{1}{2}\yv^T \yv + \frac{1}{2}\thetav^T \thetav - \yv^T \sum_{k = 1}^d \xv_k \theta_k
\end{align*}
$^{(*)}$ $\Xmat \thetav = \xv_1 \theta_1 + \xv_2 \theta_2 + \cdots + \xv_d \theta_d = \sum_{k = 1}^d \xv_k \theta_k$
\end{framei}


\begin{framei}{example: linear regression}
\item Exact CD update in direction $j$:
$$\frac{\partial{g}(\thetav)}{\partial \theta_j} = \theta_j - \yv^T \xv_j$$
\item By solving $\frac{\partial{g}(\thetav)}{\partial \theta_j} = 0$ we get $$\theta_j^\ast = \yv^T \xv_j$$
\item Repeat this update for all $\theta_j$
\end{framei}


\begin{framei}{example: soft thresholding}
\item Minimize LM with L2 loss + L1 regularization via CD (case 3):
$$\min_{\thetav} h(\thetav) = \min_{\thetav} \frac{1}{2}||\yv - \Xmat \thetav||^2 + \lambda ||\thetav||_1$$
\item Again we assume orthogonal design, i.e., $\Xmat^T \Xmat = \id$, and can write
$$h(\thetav) = \frac{1}{2}\yv^T \yv + \frac{1}{2}\thetav^T \thetav - \sum_{k = 1}^d \yv^T \xv_k \theta_k + \lambda \sum_{k = 1}^d |\theta_k|$$
\item Since $|\cdot|$ not differentiable at $0$, distinguish three cases (see next slide)
\end{framei}


\begin{framei}[fs=small,sep=L]{example: soft thresholding}
\item \textbf{Case 1:} $\theta_j > 0$.
Then $|\theta_j| = \theta_j$ and
$$0 = \frac{\partial{h}(\thetav)}{\partial \theta_j} = \theta_j - \yv^T \xv_j + \lambda \quad \Leftrightarrow \quad \theta^\ast_{j, \text{Lasso}} = \theta^\ast_j - \lambda$$
Consistent with $\theta_j > 0$ only if $\theta^\ast_j > \lambda$
\item \textbf{Case 2:} $\theta_j < 0$.
Then $|\theta_j| = - \theta_j$ and
$$0 = \frac{\partial{h}(\thetav)}{\partial \theta_j} = \theta_j - \yv^T \xv_j - \lambda \quad \Leftrightarrow \quad \theta^\ast_{j, \text{Lasso}} = \theta^\ast_j + \lambda$$
Consistent with $\theta_j < 0$ only if $\theta^\ast_j < -\lambda$
\item \textbf{Case 3:} $\theta_j = 0$, remaining case where $|\theta^\ast_j| \le \lambda$

\begin{framev}{example: soft thresholding}
We can write the solution as:
$$\theta^\ast_{j, \text{Lasso}} =
\begin{cases}
\theta^\ast_j - \lambda & \text{ if } \theta^\ast_j > \lambda \\
\theta^\ast_j + \lambda & \text{ if } \theta^\ast_j < - \lambda \\
0 &  \text{ if }  \theta^\ast_j \in [- \lambda, \lambda]
\end{cases}$$

This operation is called soft thresholding: Coefs for which solution to unregularized problem is not larger than threshold ($|\theta^\ast_j| \le \lambda$) are shrunk to zero\\
\spacer
{\footnotesize NB: Rigorous derivation of soft thresholding operator not trivial (subgradients)}
\end{framev}


\begin{framev}{CD for statistics and ML}
Why is it being used?
\begin{itemize}
\item Easy to implement
\item Scalable: no storage/operations on large objects,
just current point \\
\comment[NOTE]{BB}{es wäre schön die effizienz von CD gegen GD oder anderes zu zeigen. das könnte auch in einem extra chucnk mal vergleichen.
so large scale betrachtungen}
$\Rightarrow$ Good implementation can achieve state-of-the-art performance
\item Applicable for non-differentiable (but convex separable) objectives
\end{itemize}
\spacer
Examples:
\begin{itemize}
\item Lasso regression, Lasso GLM, graphical Lasso

\item Support Vector Machines
\item Regression with non-convex penalties
\end{itemize}
\spacer
Limitations:
\begin{itemize}
\item Slow (zig-zagging) if coordinates strongly correlated (title figure)
\end{itemize}
\end{framev}
